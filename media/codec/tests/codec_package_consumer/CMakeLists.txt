cmake_minimum_required(VERSION 3.20)
set(TEST_CODEC_COMPONENTS "AudioCodec;VideoCodec" CACHE STRING "Codec components to exercise")
set(codec_consumer_languages C)
if("VideoCodec" IN_LIST TEST_CODEC_COMPONENTS)
  list(APPEND codec_consumer_languages CXX)
endif()
project(TurboMediaInstalledCodecConsumer LANGUAGES ${codec_consumer_languages})
find_package(TurboMedia CONFIG REQUIRED COMPONENTS ${TEST_CODEC_COMPONENTS})
add_executable(tm_installed_codec codec_smoke.c)
target_compile_features(tm_installed_codec PRIVATE c_std_11)
foreach(component IN LISTS TEST_CODEC_COMPONENTS)
  target_compile_definitions(tm_installed_codec PRIVATE TEST_${component})
  target_link_libraries(tm_installed_codec PRIVATE TurboMedia::${component})
endforeach()
enable_testing()
add_test(NAME installed_codec_components COMMAND tm_installed_codec)
