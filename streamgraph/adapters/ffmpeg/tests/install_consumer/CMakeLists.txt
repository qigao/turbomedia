cmake_minimum_required(VERSION 3.20)
project(StreamGraphInstalledFFmpegDecoder LANGUAGES C CXX)
find_package(TurboMediaStreamGraphFFmpeg CONFIG REQUIRED)
if(NOT TARGET TurboMedia::StreamGraph OR NOT TARGET TurboMedia::StreamGraphFFmpeg)
  message(FATAL_ERROR "Installed StreamGraph FFmpeg package lacks required targets")
endif()
add_executable(tm_sg_ffmpeg_consumer_c consumer.c)
target_link_libraries(tm_sg_ffmpeg_consumer_c PRIVATE TurboMedia::StreamGraphFFmpeg)
add_executable(tm_sg_ffmpeg_consumer_cpp consumer.cpp)
target_link_libraries(tm_sg_ffmpeg_consumer_cpp PRIVATE TurboMedia::StreamGraphFFmpeg)
enable_testing()
add_test(NAME installed_ffmpeg_c COMMAND tm_sg_ffmpeg_consumer_c)
add_test(NAME installed_ffmpeg_cpp COMMAND tm_sg_ffmpeg_consumer_cpp)
