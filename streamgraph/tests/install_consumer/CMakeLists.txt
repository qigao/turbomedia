cmake_minimum_required(VERSION 3.20)
project(StreamGraphInstalledConsumers LANGUAGES C CXX)
find_package(TurboMediaStreamGraph CONFIG REQUIRED)
add_executable(streamgraph_consumer_c ../install_consumer.c)
add_executable(streamgraph_consumer_cpp ../install_consumer.cpp)
target_link_libraries(streamgraph_consumer_c PRIVATE TurboMedia::StreamGraph)
target_link_libraries(streamgraph_consumer_cpp PRIVATE TurboMedia::StreamGraph)
enable_testing()
add_test(NAME installed_c COMMAND streamgraph_consumer_c)
add_test(NAME installed_cpp COMMAND streamgraph_consumer_cpp)
