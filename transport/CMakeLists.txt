file(GLOB TURBO_MEDIA_RTP_SOURCES CONFIGURE_DEPENDS rtp/*.c)
file(GLOB TURBO_MEDIA_RTP_EXT_SOURCES CONFIGURE_DEPENDS rtp/rtpext/*.c)
file(GLOB TURBO_MEDIA_RTP_PAYLOAD_SOURCES CONFIGURE_DEPENDS rtp/payload/*.c)
file(GLOB_RECURSE TURBO_MEDIA_SIP_SOURCES CONFIGURE_DEPENDS sip/*.c)

target_sources(
  turbo_media_transport
  PRIVATE ${TURBO_MEDIA_RTP_SOURCES}
          ${TURBO_MEDIA_RTP_EXT_SOURCES}
          ${TURBO_MEDIA_RTP_PAYLOAD_SOURCES}
          ${TURBO_MEDIA_SIP_SOURCES})
target_include_directories(
  turbo_media_transport
  PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/rtp>
         $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/rtp/include>
         $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/rtp/rtpext>
         $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/rtp/payload>
  PRIVATE sip sip/include)
target_compile_definitions(
  turbo_media_transport
  PUBLIC TURBO_MEDIA_HAS_RTP TURBO_MEDIA_HAS_SIP)
