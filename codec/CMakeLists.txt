target_sources(turbo_media PRIVATE
  ${CMAKE_CURRENT_LIST_DIR}/codec_registry.c
  ${CMAKE_CURRENT_LIST_DIR}/g711_codec.c)

if(TURBO_MEDIA_ENABLE_OPUS)
  find_path(OPUS_INCLUDE_DIR NAMES opus/opus.h REQUIRED)
  find_library(OPUS_LIBRARY NAMES opus REQUIRED)
  target_sources(turbo_media PRIVATE ${CMAKE_CURRENT_LIST_DIR}/opus_codec.c)
  target_include_directories(turbo_media PRIVATE ${OPUS_INCLUDE_DIR})
  target_link_libraries(turbo_media PRIVATE ${OPUS_LIBRARY})
  target_compile_definitions(turbo_media PUBLIC TURBO_MEDIA_HAS_OPUS)
endif()
